```java
package com.xune.cosguard;

import java.util.Collections;
import java.util.LinkedHashSet;
import java.util.Set;

/** guard.conf 解析结果(不可变)。parse 任何异常折叠为 invalid, 调用方按 degraded 处理。 */
public final class GuardConfig {
    public static final String DEFAULT_BLOCK = "o-stop,frozen,cached,empty,cpu";
    public static final int VERSION = 1;

    public final String mode;              // "observe" | "guard"
    public final Set<String> blockRules;   // 有序
    public final Set<String> white;
    public final boolean valid;
    public final String error;             // valid 时为 ""

    private GuardConfig(String mode, Set<String> block, Set<String> white,
                        boolean valid, String error) {
        this.mode = mode; this.blockRules = block; this.white = white;
        this.valid = valid; this.error = error;
    }

    public static GuardConfig parse(String content) {
        try {
            if (content == null || content.trim().isEmpty()) return invalid("empty");
            String mode = "observe";
            Set<String> block = splitSet(DEFAULT_BLOCK);
            Set<String> white = new LinkedHashSet<>();
            boolean ver = false;
            for (String raw : content.split("\n")) {
                String line = raw.trim();
                if (line.isEmpty() || line.startsWith("#")) continue;
                if (line.startsWith("VERSION="))
                    ver = Integer.parseInt(line.substring(8).trim()) == VERSION;
                else if (line.startsWith("MODE=")) {
                    String m = line.substring(5).trim();
                    if ("guard".equals(m) || "observe".equals(m)) mode = m;
                } else if (line.startsWith("BLOCK="))
                    block = splitSet(line.substring(6).trim());
                else if (line.startsWith("WHITE ")) {
                    String p = line.substring(6).trim();
                    if (p.matches("[a-zA-Z][a-zA-Z0-9._]+")) white.add(p);
                }
            }
            if (!ver) return invalid("version");
            if (white.isEmpty()) return invalid("no-white");      // S2
            return new GuardConfig(mode, Collections.unmodifiableSet(block),
                    Collections.unmodifiableSet(white), true, "");
        } catch (Throwable t) {
            return invalid("parse:" + t);
        }
    }

    private static GuardConfig invalid(String err) {
        return new GuardConfig("observe",
                Collections.unmodifiableSet(splitSet(DEFAULT_BLOCK)),
                Collections.unmodifiableSet(new LinkedHashSet<>()), false, err);
    }

    private static Set<String> splitSet(String csv) {
        Set<String> s = new LinkedHashSet<>();
        for (String p : csv.split(",")) { String t = p.trim(); if (!t.isEmpty()) s.add(t); }
        return s;
    }
}
```

```java
package com.xune.cosguard;

import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

/**
 * 拦截决策链 (spec §6.2): WHITE → MODE → BLOCK → dedup → FUSE → BLOCK.
 * 纯逻辑无 IO; null=放行不记; act==BLOCK 调用方 skip killLocked.
 * 状态(fuse/dedup)按 pid 隔离, 自带剪枝防膨胀。
 */
public final class GuardEngine {
    public enum Act { BLOCK, PASS_OBSERVE, PASS_NO_RULE, PASS_DUP, FUSE, ERROR }

    public static final class Decision {
        public final Act act;
        public final String rule;
        public Decision(Act act, String rule) { this.act = act; this.rule = rule; }
    }

    private static final int FUSE_THRESHOLD = 10;   // spec: 连续拦满10次放行1次
    private static final long DEDUP_MS = 500;       // 500ms 去重窗: 挡同栈重载互调(µs级), 放行>=500ms真实重试
    private static final long PRUNE_MS = 60_000;

    private final Map<String, Integer> fuse = new ConcurrentHashMap<>();   // pid:rule -> count
    private final Map<String, Long> dedup = new ConcurrentHashMap<>();     // pid:rule:reason -> lastMs
    private volatile long lastPrune = 0;

    /** 规则名 → reason 匹配 (spec §5.2)。无对应规则的杀因(remove task等)永不可拦。 */
    public static String matchRule(GuardConfig cfg, String reason) {
        if (reason == null) return null;
        for (String r : cfg.blockRules) {
            switch (r) {
                case "o-stop": if (reason.contains("due to o-stop")) return r; break;
                case "frozen": if (reason.contains("frozen")) return r; break;
                case "cached": if (reason.startsWith("cached #")) return r; break;
                case "empty":  if (reason.startsWith("empty "))  return r; break;
                case "cpu":    if (reason.contains("excessive cpu")) return r; break;
                default: break;   // 未知规则名: 匹配不到, 无害 (S3)
            }
        }
        return null;
    }

    /** 返回 null=放行不记; 非null=必须记遥测; BLOCK=拦截(不调 proceed)。 */
    public Decision decide(GuardConfig cfg, String pkg, String reason,
                           long pid, long nowMs) {
        prune(nowMs);
        // step2: 白名单外零开销直通(不记)
        if (pkg == null || !cfg.white.contains(pkg)) return null;
        // step3: observe/degraded 模式只记不拦 (ConfigStore 降级时强制 observe)
        if (!"guard".equals(cfg.mode)) return new Decision(Act.PASS_OBSERVE, "");
        // step4: 规则匹配
        String rule = matchRule(cfg, reason);
        if (rule == null) return new Decision(Act.PASS_NO_RULE, "");
        // step5: 500ms 去重(同栈多重载互调只算一次; >=500ms 的系统重试视为新事件计入 fuse)
        String dKey = pid + ":" + rule + ":" + (reason == null ? "" : reason);
        Long last = dedup.get(dKey);
        if (last != null && nowMs - last < DEDUP_MS) return new Decision(Act.PASS_DUP, rule);
        dedup.put(dKey, nowMs);
        // step6: 保险丝
        String fKey = pid + ":" + rule;
        int n = fuse.getOrDefault(fKey, 0);
        if (n >= FUSE_THRESHOLD) {
            fuse.remove(fKey);                       // 放行1次并清零, 让系统收敛 (S4)
            return new Decision(Act.FUSE, rule);
        }
        // step7: 拦截
        fuse.put(fKey, n + 1);
        return new Decision(Act.BLOCK, rule);
    }

    /** 防膨胀: 60s 清一次过期 dedup 条目; fuse 只在放行时清, 不做时间衰减(需保证收敛)。 */
    private void prune(long nowMs) {
        if (nowMs - lastPrune < PRUNE_MS) return;
        lastPrune = nowMs;
        dedup.entrySet().removeIf(en -> nowMs - en.getValue() > DEDUP_MS * 4);
    }
}
```

```java
package com.xune.cosguard;

import java.io.File;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;

/**
 * guard.conf 热加载 (spec §5.3): 5s 检查缓存 + mtime 变更重读.
 * 失败降级(S1): 保留 lastGood 内容但强制 mode=observe; 无 lastGood 则空名单.
 */
public final class ConfigStore {
    private static final long CHECK_MS = 5_000;
    private static final long RETRY_MS = 30_000;

    private final File file;
    private volatile GuardConfig lastGood = null;
    private volatile GuardConfig current;
    private volatile long nextCheckAt = 0;
    private volatile long lastMtime = Long.MIN_VALUE;

    public ConfigStore(File file) {
        this.file = file;
        this.current = GuardConfig.invalidOf("not-loaded");
    }

    public synchronized GuardConfig current(long nowMs) {
        if (nowMs < nextCheckAt) return current;          // 5s 缓存(失败态 30s 才重试)
        nextCheckAt = nowMs + CHECK_MS;
        long mtime;
        try { mtime = file.canRead() ? file.lastModified() : -1L; }
        catch (Throwable t) { mtime = -1L; }
        if (mtime < 0) { degrade("missing", nowMs); return current; }    // S1
        if (mtime == lastMtime && current.valid) return current;        // 未变更
        try {
            GuardConfig c = GuardConfig.parse(
                new String(Files.readAllBytes(file.toPath()), StandardCharsets.UTF_8));
            if (c.valid) {
                lastGood = c; lastMtime = mtime; current = c;
            } else {
                degrade(c.error, nowMs);                                 // S2/S3
            }
        } catch (Throwable t) {
            degrade("io:" + t, nowMs);
        }
        return current;
    }

    /** 统一用调用方 nowMs 推进重试点, 保证逻辑时钟一致(可测)。 */
    private void degrade(String why, long nowMs) {
        nextCheckAt = nowMs + RETRY_MS;                  // S1: 失败态 30s 后重试
        current = (lastGood != null) ? GuardConfig.degradedCopy(lastGood, why)
                                     : GuardConfig.invalidOf(why);
    }
}
```

```java
public final class Telemetry {
    private final File file;
    public Telemetry(File file) { this.file = file; }

    public void write(int pid, String pkg, String rule, GuardEngine.Act act, String reason) {
        append(pid, pkg, rule, act.name(), reason);
    }
    public void configError(String reason) {
        append(0, "*", "config", GuardEngine.Act.ERROR.name(), reason);

    private void append(int pid, String pkg, String rule, String act, String reason) {
        try {
            String line = (System.currentTimeMillis() / 1000) + "|" + pid + "|"
                    + safe(pkg) + "|" + safe(rule) + "|" + act + "|"
                    + (reason == null ? "" : reason.replace('|', '/')) + "\n";
            Files.write(file.toPath(), line.getBytes(StandardCharsets.UTF_8),
                    StandardOpenOption.CREATE, StandardOpenOption.APPEND);
        } catch (Throwable ignored) { /* fail-open */ }
    }
    private static String safe(String s) { return s == null ? "" : s.replace('|', '/'); }
}
```

```java
        install(param.getClassLoader());

    @Override
    public void onPackageReady(PackageReadyParam param) {
        String pkg = param.getPackageName();
        if ("android".equals(pkg) || "system".equals(pkg)) {
            log(android.util.Log.WARN, "COSGuard", "entry: onPackageReady pkg=" + pkg);
            install(param.getClassLoader());
        }
    }

    private void install(ClassLoader cl) {
        if (cl == null || !installed.compareAndSet(false, true)) return;
        int hooked = 0;
        try {
            Class<?> c = Class.forName("com.android.server.am.ProcessRecord", false, cl);
            for (Method m : c.getDeclaredMethods()) {
                if (!"killLocked".equals(m.getName()) || m.getParameterCount() < 5) continue;
                hook(m).intercept(new KillHooker());
                hooked++;
            }
        } catch (Throwable t) {
            log(android.util.Log.WARN, "COSGuard", "hook install failed: " + t);
        }
        log(android.util.Log.WARN, "COSGuard",
                "COSGuard loaded, killLocked hooks=" + hooked);
```

```java
    private final class KillHooker implements Hooker {
        @Override
        public Object intercept(Chain chain) throws Throwable {
            try {
                long now = System.currentTimeMillis();
                GuardConfig cfg = store.current(now);
                maybeLogConfigError(cfg);
                Object app = chain.getThisObject();          // killLocked 是实例方法, this=ProcessRecord
                String pkg = pkgOf(app);
                int pid = pidOf(app);
                String reason = String.valueOf(chain.getArgs().get(0));
                GuardEngine.Decision d = engine.decide(cfg, pkg, reason, pid, now);
                if (d != null) {
                    telemetry.write(pid, pkg, d.rule, d.act, reason);
                    if (d.act == GuardEngine.Act.BLOCK) {
                        log(android.util.Log.WARN, "COSGuard",
                                "BLOCK pid=" + pid + " pkg=" + pkg + " rule=" + d.rule);
                        return null;                         // ★ skip: 不调 proceed, 进程不死
                    }
                }
            } catch (Throwable t) {
                try { telemetry.write(0, "*", "hook", GuardEngine.Act.ERROR, String.valueOf(t)); }
                catch (Throwable ignored) { }
            }
            return chain.proceed();                          // fail-open 兜底
        }
    }

    private void maybeLogConfigError(GuardConfig cfg) {
        if (!cfg.valid) {
            if (!configErrorLogged) { configErrorLogged = true; telemetry.configError(cfg.error); }
        } else configErrorLogged = false;
    }

    private static String pkgOf(Object app) {                // public ApplicationInfo info
        try {
            Field f = app.getClass().getField("info");
            return (String) f.get(app).getClass().getField("packageName").get(f.get(app));
        } catch (Throwable t) { return null; }               // null → decide 放行不记
    }

    private static int pidOf(Object app) {                   // public final int pid
        try { return app.getClass().getField("pid").getInt(app); }
        catch (Throwable t) { return -1; }
    }
}
```

### Task 8: guard_stats.sh（遥测聚合 → JSON）

**Files:**
- Create: `engine/guard_stats.sh`、`tests/test_guard_stats.sh`

**Interfaces:**
- Consumes: 遥测行 `TS|PID|PKG|RULE|ACT|REASON`；归档 `data/guard/YYYY-MM-DD.log`（按天分文件）
- Produces: `sh engine/guard_stats.sh [YYYY-MM-DD]` → stdout 单个 JSON。字段（spec §7.1 执行版）：`date, mode, module(installed|absent), live_today, snapshot_age_s, fuse_threshold, today{block,fuse,error,pass_observe}, hourly[24]{h,block,fuse}, rules[]{rule,n}, whitelist[]{pkg,block,last_ts,last_act}, recent[10]{ts,pkg,rule,act,reason}, dates[]`
- **与 spec 的两处注明差异**：①`whitelist` 无 `name` 字段——中文名由 W4 组件前端映射表补（数据与展示分离）②`recent` 不含原始 `TS|PID` 之外多余字段，`reason` 经转义
- 测试注入（环境变量）：`GS_BRIDGE / GS_LIVE / GS_ARCHIVE / GS_MOD`
- 实现约定：日期过滤靠"归档按天 + live 仅当天并入"（busybox awk 无 strftime）；hourly 按 **UTC+8 固定偏移**分桶（代码注明）；JSON 由 awk 一次拼出，shell 只注入 `dates[]` 尾段

```sh
BODY=$(awk -F'|' -v date="$DATE" -v mode="$MODE" -v module="$MODULE" \
           -v age="$AGE" -v lt="$LIVE_TODAY" -v wl="$WL" '
function jesc(s){ gsub(/\\/,"\\\\",s); gsub(/"/,"\\\"",s); gsub(/\r/,"",s); return s }
$1 ~ /^[0-9]+$/ && NF >= 6 {
  act = $5
  cnt[act]++
  if (act == "BLOCK") {
    if (!($4 in rules)) rules[$4] = 0
    rules[$4]++
    hb[int(($1 + 28800) % 86400 / 3600)]++      # 设备固定 UTC+8 按本地小时分桶
    pblock[$3]++
  }
  if (act == "FUSE") hf[int(($1 + 28800) % 86400 / 3600)]++
  if ($2 != "0" && index(" " wl " ", " "$3 " ") > 0) { last[$3]=$1; lastact[$3]=act }
  n++
  if (n <= 200) { rts[n]=$1; rpkg[n]=$3; rrule[n]=$4; ract[n]=act; rreason[n]=$6 }
}
END {
  printf "{\"date\":\"%s\",\"mode\":\"%s\",\"module\":\"%s\",\"live_today\":%s,", \
         date, mode, module, (lt ? "true" : "false")
  printf "\"snapshot_age_s\":%d,\"fuse_threshold\":10,", age
  printf "\"today\":{\"block\":%d,\"fuse\":%d,\"error\":%d,\"pass_observe\":%d},", \
         cnt["BLOCK"]+0, cnt["FUSE"]+0, cnt["ERROR"]+0, cnt["PASS_OBSERVE"]+0
  printf "\"hourly\":["
  for (h = 0; h < 24; h++)
    printf "%s{\"h\":%d,\"block\":%d,\"fuse\":%d}", (h ? "," : ""), h, hb[h]+0, hf[h]+0
  printf "],\"rules\":["
  f = 1
  for (k in rules) { printf "%s{\"rule\":\"%s\",\"n\":%d}", (f ? "" : ","), k, rules[k]; f = 0 }
  printf "],\"whitelist\":["
  f = 1
  for (k in last) {
    printf "%s{\"pkg\":\"%s\",\"block\":%d,\"last_ts\":%d,\"last_act\":\"%s\"}", \
           (f ? "" : ","), k, pblock[k]+0, last[k], lastact[k]
    f = 0
  }
  printf "],\"recent\":["
  f = 1; shown = 0
  for (i = n; i >= 1 && shown < 10; i--) {
    if (i > 200 || rts[i] == "") continue
    printf "%s{\"ts\":%d,\"pkg\":\"%s\",\"rule\":\"%s\",\"act\":\"%s\",\"reason\":\"%s\"}", \
           (f ? "" : ","), rts[i], rpkg[i], rrule[i], ract[i], jesc(rreason[i])
    f = 0; shown++
  }
  printf "]"
}' "$SRC")

printf '%s,"dates":[%s]}\n' "$BODY" "$DATES"
rm -f "$SRC"
```

```ts
import { ref } from 'vue'

export interface GuardStats {
  date: string; mode: string; module: string; live_today: boolean
  snapshot_age_s: number; fuse_threshold: number
  today: { block: number; fuse: number; error: number; pass_observe: number }
  hourly: { h: number; block: number; fuse: number }[]
  rules: { rule: string; n: number }[]
  whitelist: { pkg: string; block: number; last_ts: number; last_act: string }[]
  recent: { ts: number; pkg: string; rule: string; act: string; reason: string }[]
  dates: string[]
}

export const GUARD_NAME: Record<string, string> = {
  'com.tencent.mm': '微信',
  'com.tencent.mobileqq': 'QQ',
  'com.ss.android.ugc.aweme': '抖音',
}
export function guardName(pkg: string): string {
  return GUARD_NAME[pkg] ?? pkg.split('.').pop() ?? pkg
}

const emptyHourly = () => Array.from({ length: 24 }, (_, h) => ({ h, block: 0, fuse: 0 }))

export function parseGuard(json: string): GuardStats | null {
  try {
    const o = JSON.parse(json)
    if (!o || typeof o !== 'object') return null
    return {
      date: String(o.date ?? ''), mode: String(o.mode ?? 'observe'),
      module: String(o.module ?? 'absent'), live_today: !!o.live_today,
      snapshot_age_s: Number(o.snapshot_age_s ?? -1),
      fuse_threshold: Number(o.fuse_threshold ?? 10),
      today: {
        block: Number(o.today?.block ?? 0), fuse: Number(o.today?.fuse ?? 0),
        error: Number(o.today?.error ?? 0), pass_observe: Number(o.today?.pass_observe ?? 0),
      },
      hourly: Array.isArray(o.hourly) && o.hourly.length === 24 ? o.hourly : emptyHourly(),
      rules: Array.isArray(o.rules) ? o.rules : [],
      whitelist: Array.isArray(o.whitelist) ? o.whitelist : [],
      recent: Array.isArray(o.recent) ? o.recent : [],
      dates: Array.isArray(o.dates) ? o.dates.map(String) : [],
    }
  } catch { return null }
}

export const guardDate = ref<string>('')   // 空 = 今天
```
