## Global Constraints

- conf 默认 `/sdcard/Android/COSMemory/名单列表.conf`, `LIST_PATH` 环境变量覆盖 (测试注入点)。
- 前端正则与 engine/lists.sh awk 同源: PKG=`^[a-zA-Z][a-zA-Z0-9._]+$`, PKG_SUB=`^[a-zA-Z][a-zA-Z0-9._]+:.+$`, FREEZE=`PKG|PKG_SUB` (带可选 `:group`)。
- 写路径 fail-safe: 校验不过/parse 验证不过/超行数上限一律不落盘 (tmp 丢弃), 输出 `ERR:<code>`。
- ERR 码: `badfmt|dup|conflict|parsefail|iofail` (+补充 `nomatch`=del 未命中, `toomany`=超 500 行); 成功 `OK`; guard_bridge 失败输出 `OK`+`WARN:bridge` (conf 不回滚)。
- target 字符白名单 `[a-zA-Z0-9._:]`, 长度 ≤128; conf 新增后总行数 ≤500。
- 设备 shell 纪律: grep 用 -E 不用 `\|`; 测试后台必须 `>/dev/null 2>&1`; 进程级操作脚本文件化。
- 构建: panel `npx vite build`; cosguard 带 `-Pandroid.aapt2FromMavenOverride=/workspace/tools/aapt2-qemu/aapt2`。
- 部署: Linux cp → /storage/emulated/0/Download/ 中转 → Android cp → 目标, md5 双端核对。
- 版本: 模块 v0.7.0/versionCode 7000; APK 接续现版本随 Task 8 升。署名 README "- 是你吗薰儿"。

```sh
#!/system/bin/sh
# engine/bridge.sh — guard_bridge 抽取 (自 service.sh, DRY: service/listedit 双 source)
# 依赖环境: MODDIR, BRIDGE_DIR, BRIDGE, parse_lists (engine/lists.sh 由调用方 source)
guard_bridge() {
  mkdir -p "$BRIDGE_DIR" 2>/dev/null
  chown system:system "$BRIDGE_DIR" 2>/dev/null
  chmod 775 "$BRIDGE_DIR" 2>/dev/null
  local jc="$MODDIR/config/memory.json" mode="" block="" fze="" fzline=""
  if [ -f "$jc" ]; then
    mode=$(sed -n 's/.*"mode" *: *"\([^"]*\)".*/\1/p' "$jc" | head -1)
    block=$(sed -n 's/.*"block_rules" *: *"\([^"]*\)".*/\1/p' "$jc" | head -1)
    fzline=$(sed -n "/\"freeze\"/,/}/p" "$jc" | grep "enabled" | head -1)
  fi
  case "$fzline" in *true*) fze=1;; *) fze=0;; esac
  [ "$mode" = guard ] || mode=observe
  [ -n "$block" ] || block=o-stop,frozen,cached,empty,cpu
  eval "$(parse_lists "${LIST_PATH:-/sdcard/Android/COSMemory/名单列表.conf}" 2>/dev/null)"
  {
    echo "# COSGuard bridge - generated by COSMemory service.sh"
    echo "VERSION=1"
    echo "MODE=$mode"
    echo "BLOCK=$block"
    for p in $WHITE_LIST; do echo "WHITE $p"; done
    if [ "$fze" = 1 ] && [ -n "$FREEZE_LIST" ]; then
      echo "FREEZE_ENABLED=1"
      for p in $FREEZE_LIST; do echo "FREEZE $p"; done
    fi
  } > "$BRIDGE.tmp" 2>/dev/null && mv -f "$BRIDGE.tmp" "$BRIDGE" 2>/dev/null \
    && chmod 644 "$BRIDGE" && chown system:system "$BRIDGE"
}
```

#!/system/bin/sh
# engine/listedit.sh — 名单编辑写入层 (spec list-editor §4.2)
# 用法: sh listedit.sh add|del <WHITE|KILL|FREEZE> <target>
MODDIR="${MODDIR:-/data/adb/modules/COSMemory}"
CONF="${LIST_PATH:-/sdcard/Android/COSMemory/名单列表.conf}"
BRIDGE_DIR="${BRIDGE_DIR:-/data/system/cosmem}"
BRIDGE="${BRIDGE:-$BRIDGE_DIR/guard.conf}"
. "$MODDIR/engine/lists.sh"
. "$MODDIR/engine/bridge.sh"
PKG='^[a-zA-Z][a-zA-Z0-9._]+$'
PKG_SUB='^[a-zA-Z][a-zA-Z0-9._]+:.+$'
MAX_LINES=500

validate() { # $1=group $2=target → stdout=errcode rc=0/1
  g="$1"; t="$2"
  [ -n "$g" ] && [ -n "$t" ] || { echo badfmt; return 1; }
  [ ${#t} -le 128 ] || { echo badfmt; return 1; }
  case "$g" in
    WHITE)  printf '%s' "$t" | grep -qE "$PKG" || { echo badfmt; return 1; };;
    KILL)   printf '%s' "$t" | grep -qE "$PKG_SUB" || { echo badfmt; return 1; };;
    FREEZE) printf '%s' "$t" | grep -qE "$PKG|$PKG_SUB" || { echo badfmt; return 1; };;
    *) echo badfmt; return 1;;
  esac
  return 0
}
conf_stats() { # $1=file → "bad_count conflict_count"
  eval "$(parse_lists "$1" 2>/dev/null)"
  echo "$LIST_BAD $(printf '%s' "$FREEZE_CONFLICT" | wc -w)"
}

```ts
/** 对齐 awk FREEZE 规则: pkg 或 pkg:group (spec list-editor §4.1) */
export const FREEZE_FULL = /^[a-zA-Z][a-zA-Z0-9._]+(:.+)?$/
export type ListGroup = 'white' | 'kill' | 'freeze'
export type EditErr = '' | 'badfmt' | 'dup' | 'conflict'

/**
 * 编辑校验: 格式 → 重复 → D4 冲突 (包名级, 比 awk 整串更严 — 前端友好拦截)。
 * dup 语义分组: kill 按整串 (同包不同进程后缀是不同目标), white/freeze 按剥组包名。
 */
export function validateEdit(group: ListGroup, target: string, conf: ListConf): EditErr {
  const re = group === 'kill' ? PKG_SUB : group === 'freeze' ? FREEZE_FULL : PKG
  if (!re.test(target)) return 'badfmt'
  const pkg = target.split(':')[0]
  if (group === 'kill') {
    if (conf.kill.includes(target)) return 'dup'
    return ''
  }
  if (conf[group].some(o => o.split(':')[0] === pkg)) return 'dup'
  const other = group === 'freeze' ? conf.white : conf.freeze
  if (other.some(o => o.split(':')[0] === pkg)) return 'conflict'
  return ''
}

/** 表单三字段 → conf target (KILL 后缀必填由 validateEdit 把关) */
export function buildTarget(group: ListGroup, pkg: string, suffix: string): string {
  const s = suffix.replace(/^:+/, '')
  if (group === 'kill') return `${pkg}:${s}`
```

```ts
import { execRead, type KsuBridge } from './ksu'

export type EditOp = 'add' | 'del'
export type EditGroup = 'WHITE' | 'KILL' | 'FREEZE'
export interface EditResult { ok: boolean; err: string }

const SAFE = /^[a-zA-Z0-9._:]+$/   // 白名单防注入 (与 listedit.sh validate 同集)

export async function listEdit(
  bridge: KsuBridge, mod: string, op: EditOp, group: EditGroup, target: string
): Promise<EditResult> {
  if (!SAFE.test(target)) return { ok: false, err: 'ERR:badfmt' }
  try {
    const out = await execRead(bridge, `sh ${mod}/engine/listedit.sh ${op} ${group} ${target}`)
    const first = out.split('\n')[0].trim()
    if (first === 'OK') return { ok: true, err: '' }
    return { ok: false, err: first || 'ERR:empty' }
  } catch {
    return { ok: false, err: 'ERR:empty' }
  }
}
```

```java
package com.xune.cosguard;
import java.util.*;
/** apps.json 纯逻辑 (spec list-editor §4.4) — 无 Android 依赖, JUnit 可测。 */
public final class CatalogCore {
    public static final class App {
        public final String pkg, label; public final boolean third;
        public App(String pkg, String label, boolean third) {
            this.pkg = pkg; this.label = label; this.third = third;
        }
    }
    public static String toJson(List<App> apps) {
        List<App> sorted = new ArrayList<>(apps);
        sorted.sort(Comparator.comparing(a -> a.pkg));
        StringBuilder sb = new StringBuilder("{\"v\":1,\"apps\":[");
        for (App a : sorted) {
            if (sb.length() > 11) sb.append(',');
            sb.append("{\"p\":\"").append(esc(a.pkg))
              .append("\",\"l\":\"").append(esc(a.label))
              .append("\",\"t\":").append(a.third ? 1 : 0).append('}');
        }
        return sb.append("]}").toString();
    }
    private static String esc(String s) {
        StringBuilder sb = new StringBuilder(s.length() + 8);
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c == '"' || c == '\\') sb.append('\\').append(c);
            else if (c < 0x20) sb.append(String.format("\\u%04x", (int) c));
            else sb.append(c);
        }
        return sb.toString();
    }
    public static boolean shouldRebuild(long now, long jsonMtime, long refreshMtime, long lastSeenRefresh) {
        if (jsonMtime < 0) return true;
        if (refreshMtime > lastSeenRefresh) return true;
        return now - jsonMtime > 86_400_000L;
    }
}
```

```java
    public void start() {

        ScheduledExecutorService es = Executors.newSingleThreadScheduledExecutor(r -> {
            Thread t = new Thread(r, "cosmem-appcatalog"); t.setDaemon(true); return t;
        });
        // 首建延迟 FIRST_MS (开机高峰后), 之后每 CHECK_MS 检查 refresh/mtime

            long now = System.currentTimeMillis();
            long jm = JSON.canRead() ? JSON.lastModified() : -1L;
            long rm = REFRESH.canRead() ? REFRESH.lastModified() : 0L;
            if (CatalogCore.shouldRebuild(now, jm, rm, lastRefreshMtime)) rebuild();
        } catch (Throwable t) {
            android.util.Log.w("COSGuard", "AppCatalog tick: " + t);
        }
    }
```

```java
    void rebuild() throws Exception {
        PackageManager pm = ctx.getPackageManager();
        List<PackageInfo> ps = pm.getInstalledPackages(0);
        List<CatalogCore.App> apps = new ArrayList<>(ps.size());
        for (PackageInfo pi : ps) {
            boolean third = (pi.applicationInfo.flags & ApplicationInfo.FLAG_SYSTEM) == 0;
            String label;
            try { label = String.valueOf(pm.getApplicationLabel(pi.applicationInfo)); }
            catch (Throwable t) { label = pi.packageName; }
            apps.add(new CatalogCore.App(pi.packageName, label, third));
            if (third) writeIcon(pm, pi);
        }
        byte[] body = CatalogCore.toJson(apps).getBytes(StandardCharsets.UTF_8);
        File tmp = new File(DIR, "apps.json.tmp");
        Files.write(tmp.toPath(), body);
        if (!tmp.renameTo(JSON)) throw new IOException("rename apps.json");
        JSON.setReadable(true, false);
        lastRefreshMtime = REFRESH.canRead() ? REFRESH.lastModified() : 0L;
    }
    private void writeIcon(PackageManager pm, PackageInfo pi) {
        try {
            Drawable d = pm.getApplicationIcon(pi.applicationInfo);
            int w = d.getIntrinsicWidth(), h = d.getIntrinsicHeight();
            if (w <= 0 || h <= 0) return;
            Bitmap bmp = Bitmap.createBitmap(Math.min(w, 128), Math.min(h, 128),
                    Bitmap.Config.ARGB_8888);
            Canvas cv = new Canvas(bmp);
            d.setBounds(0, 0, bmp.getWidth(), bmp.getHeight());
            d.draw(cv);
            File tmp = new File(ICON_DIR, pi.packageName + ".tmp");
            try (FileOutputStream fo = new FileOutputStream(tmp)) {
                bmp.compress(Bitmap.CompressFormat.PNG, 100, fo);
            }
            bmp.recycle();
            File out = new File(ICON_DIR, pi.packageName + ".png");
            if (!tmp.renameTo(out)) out.delete();
        } catch (Throwable ignored) { /* 单包 icon 失败不阻断整体 */ }
    }
}
```

```ts
export interface AppEntry { p: string; l: string; t: number }
export function parseApps(json: string): AppEntry[] {
  try {
    const o = JSON.parse(json)
    if (!o || o.v !== 1 || !Array.isArray(o.apps)) return []
    return o.apps.filter((a: any) => a && typeof a.p === 'string' && typeof a.l === 'string')
      .map((a: any) => ({ p: a.p, l: a.l, t: a.t ? 1 : 0 }))
  } catch { return [] }
}
export function filterApps(apps: AppEntry[], q: string): AppEntry[] {
  const s = q.trim().toLowerCase()
  if (!s) return apps
  return apps.filter(a => a.l.toLowerCase().includes(s) || a.p.toLowerCase().includes(s))
}
/** 首屏 icon 批量取: 单次 exec 输出 "===<pkg>\n<base64>" 分段 (spec §4.4 ≤2 exec 约束) */
export function iconCmd(pkgs: string[]): string {
  const dir = '/data/system/cosmem/icons'
  return pkgs.map(p => `echo ====${p}; base64 -w0 ${dir}/${p}.png 2>/dev/null; echo`).join('; ')
}
export function parseIcons(out: string): Record<string, string> {
  const m: Record<string, string> = {}
  for (const seg of out.split('====').slice(1)) {
    const nl = seg.indexOf('\n')
    if (nl > 0) { const pkg = seg.slice(0, nl).trim(); const b = seg.slice(nl + 1).trim()
      if (pkg && b) m[pkg] = b }
  }
  return m
}
```

```ts
import { ref } from 'vue'
import { parseListConf, validateEdit } from '../composables/listconf'
import { detectBridge } from '../composables/ksu'
import { listEdit } from '../composables/listedit-client'

const props = defineProps<{ stats: GuardStats | null; listRaw: string; mod: string }>()
const emit = defineEmits<{ (e: 'pick', date: string): void; (e: 'edited'): void }>()
const frozen = ref<Record<string, string>>({})   // pkg → '已加' | err

async function addFreeze(pkg: string) {
  const b = detectBridge()
  if (!b) { frozen.value[pkg] = 'ERR:no-bridge'; return }
  const conf = parseListConf(props.listRaw)
  const e = validateEdit('freeze', pkg, conf)
  if (e) { frozen.value[pkg] = 'ERR:' + e; return }
  const r = await listEdit(b, props.mod, 'add', 'FREEZE', pkg)
  if (r.ok) { frozen.value[pkg] = '已加'; emit('edited') }
  else frozen.value[pkg] = r.err
}
```
